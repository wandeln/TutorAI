{% block solve_scripts %}
  // ─── CodeMirror Editor ──────────────────────────────────────
  {% if task.task_type == 'code' %}
  const editor = CodeMirror.fromTextArea(document.getElementById('code-editor'), {
    mode: { name: 'python', version: 3 },
    theme: 'oneDark',
    lineNumbers: true,
    autoCloseBrackets: true,
    indentUnit: 4,
    tabSize: 4,
    lineWrapping: true,
    matchBrackets: true,
  });
  window._codeEditor = editor;
  {% endif %}

  // ─── Run Code (Console Output + Plots) ─────────────────────
  document.getElementById("run-code-btn")?.addEventListener("click", async () => {
    const btn = document.getElementById("run-code-btn");
    const consoleDiv = document.getElementById("console-output");
    const consoleText = document.getElementById("console-text");
    const plotDiv = document.getElementById("plot-images");
    const plotGallery = document.getElementById("plot-gallery");

    const code = window._codeEditor ? window._codeEditor.getValue() : "";
    if (!code.trim()) {
      showToast("Bitte schreibe zuerst Code.", "warning");
      return;
    }

    btn.disabled = true;
    consoleDiv.classList.remove("hidden");
    plotDiv.classList.add("hidden");
    plotGallery.innerHTML = "";
    consoleText.textContent = "Code wird ausgefuhrt...";
    consoleText.style.color = "#9ca3af";

    try {
      const res = await fetch("/api/student/tasks/{{ task.id }}/run-code", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ code_solution: code }),
        credentials: "same-origin",
      });

      let data;
      try {
        data = await res.json();
      } catch {
        data = { detail: "Server-Fehler — bitte versuche es erneut." };
      }

      if (!res.ok) {
        consoleText.textContent = data.detail || "Fehler";
        consoleText.style.color = "#f87171";
        return;
      }

      // Zeige stdout (immer sichtbar, auch bei Fehlern)
      let output = data.stdout || "";
      if (data.timeout) {
        output += "\n[TIMEOUT: Code hat die Zeitueberschreitung erreicht.]"
      }
      // Fuege stderr/Anhang falls vorhanden
      if (data.stderr && data.stderr.trim()) {
        output += "\n\n--- FEHLER ---\n" + data.stderr.trim();
      }
      if (!output.trim()) {
        output = "(Keine Ausgabe)"
      }

      consoleText.textContent = output;
      consoleText.style.color = data.stderr ? "#f87171" : "#86efac";

      // Zeige Matplotlib-Bilder falls vorhanden
      if (data.images && data.images.length > 0) {
        plotDiv.classList.remove("hidden");
        data.images.forEach((img, idx) => {
          const imgEl = document.createElement("img");
          imgEl.src = "data:image/png;base64," + img.base64;
          imgEl.className = "max-w-full rounded-lg border border-gray-200 shadow-sm";
          imgEl.alt = "Plot " + (idx + 1);
          imgEl.title = "Plot " + (idx + 1) + " (Klick zum Groeanzeigen)";
          imgEl.style.cursor = "pointer";
          imgEl.onclick = () => {
            const overlay = document.createElement("div");
            overlay.className = "fixed inset-0 z-50 bg-black bg-opacity-75 flex items-center justify-center p-4";
            overlay.onclick = () => overlay.remove();
            const bigImg = document.createElement("img");
            bigImg.src = imgEl.src;
            bigImg.className = "max-w-full max-h-full rounded-lg shadow-xl";
            overlay.appendChild(bigImg);
            document.body.appendChild(overlay);
          };
          plotGallery.appendChild(imgEl);
        });
      }

    } catch (err) {
      consoleText.textContent = "Fehler: " + err.message;
      consoleText.style.color = "#f87171";
    } finally {
      btn.disabled = false;
    }
  });

  // Clear Console Button
  document.getElementById("clear-console-btn")?.addEventListener("click", () => {
    document.getElementById("console-output").classList.add("hidden");
  });

  // ─── Run Public Tests (Code Tasks) ─────────────────────────
  document.getElementById("run-tests-btn")?.addEventListener("click", async () => {
    const btn = document.getElementById("run-tests-btn");
    const output = document.getElementById("test-results");

    const code = window._codeEditor ? window._codeEditor.getValue() : "";
    if (!code.trim()) {
      showToast("Bitte schreibe zuerst Code.", "warning");
      return;
    }

    btn.disabled = true;
    output.classList.remove("hidden");
    output.innerHTML = '
⏳ Tests werden ausgeführt...
';

    try {
      const res = await fetch("/api/student/tasks/{{ task.id }}/run-tests", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ code_solution: code }),
        credentials: "same-origin",
      });

      let data;
      try {
        data = await res.json();
      } catch {
        data = { detail: "Server-Fehler — bitte versuche es erneut." };
      }

      if (!res.ok) {
        output.innerHTML = `
${data.detail || "Fehler"}
`;
        return;
      }

      // Render results
      const passed = data.tests_passed || 0;
      const total = data.tests_total || 0;
      const allPassed = total === 0 ? true : data.passed;

      let message, borderColor, textColor;
      if (total === 0) {
        message = '✅ Keine Tests vorhanden — Tests bestanden';
        borderColor = 'border-l-green-500';
        textColor = 'text-green-700';
      } else if (allPassed) {
        message = '✅ Alle Tests bestanden!';
        borderColor = 'border-l-green-500';
        textColor = 'text-green-700';
      } else {
        message = '❌ Tests nicht bestanden';
        borderColor = 'border-l-red-500';
        textColor = 'text-red-700';
      }

      let html = `
        

          

            
              ${message}
            
            ${passed}/${total} Tests
          

      `;

      data.test_results.forEach(t => {
        const icon = t.passed ? '✅' : '❌';
        const color = t.passed ? 'text-green-600' : 'text-red-600';
        let detailHtml = '';
        if (!t.passed && t.error) {
          // error kann ein String oder ein Objekt {summary, full} sein
          let summary = '';
          let fullTraceback = '';
          if (typeof t.error === 'object' && t.error.summary) {
            summary = t.error.summary;
            fullTraceback = t.error.full;
          } else {
            summary = t.error;
            fullTraceback = t.error;
          }
          detailHtml = '
'
          if (fullTraceback && fullTraceback.length > summary.length) {
            detailHtml += `
Traceback anzeigen
${escapeHtml(fullTraceback)}
`;
          }
          detailHtml += `
`;
        }
        html += `
          

            ${icon}
            

              ${t.name}
              ${detailHtml}
            

          

        `;
      });

      if (data.stderr && data.stderr.trim()) {
        html += `
Stderr: ${escapeHtml(data.stderr.trim())}
`;
      }

      html += `
`;
      output.innerHTML = html;

    } catch (err) {
      output.innerHTML = `
Fehler: ${err.message}
`;
    } finally {
      btn.disabled = false;
    }
  });
