  function wsTermNew() {
    if (typeof Terminal === "undefined" || !window.FitAddon) {
      showToast("Terminal-Bibliothek nicht geladen — Seite neu laden (Strg+Shift+R).", "error");
      return;
    }
    wsTerm.seq += 1;
    const s = { id: wsTerm.seq, label: "Terminal " + wsTerm.seq,
                ws: null, term: null, fit: null, holder: null, dead: false };
    s.holder = document.createElement("div");
    s.holder.className = "w-full h-full";
    s.holder.style.display = "none";
    document.getElementById("ws-term-host").appendChild(s.holder);
    s.term = new Terminal({
      cursorBlink: true,
      fontSize: 13,
      fontFamily: "ui-monospace, SFMono-Regular, Menlo, Consolas, monospace",
      theme: { background: "#0a0a0a" },
    });
    s.fit = new FitAddon.FitAddon();
    s.term.loadAddon(s.fit);
    s.term.open(s.holder);

    const wsProto = location.protocol === "https:" ? "wss://" : "ws://";
    s.ws = new WebSocket(wsProto + location.host + WS_API + "/terminal");
    s.ws.binaryType = "arraybuffer";
    s.ws.onopen = () => {
      try { s.ws.send(JSON.stringify({ cols: s.term.cols, rows: s.term.rows })); } catch (err) { /* ignore */ }
      if (wsTerm.active === s) s.term.focus();
    };
    s.ws.onmessage = (e) => {
      if (e.data instanceof ArrayBuffer) {
        s.term.write(new Uint8Array(e.data));
        return;
      }
      let ctrl = null;
      try { ctrl = JSON.parse(e.data); } catch (err) { s.term.write(e.data); return; }
      if (ctrl && ctrl.type === "exit") {
        s.dead = true;
        s.term.write("\r\n[Prozess beendet" + (ctrl.code != null ? " (exit " + ctrl.code + ")" : "") + "]\r\n");
        wsTermRenderList();
        wsTreeAutoRefreshUpdate();
      } else if (ctrl && ctrl.type === "error") {
        s.dead = true;
        s.term.write("\r\n[Fehler: " + (ctrl.message || "unbekannt") + "]\r\n");
        wsTermRenderList();
        wsTreeAutoRefreshUpdate();
      } else {
        s.term.write(String(e.data));
      }
    };
    s.ws.onclose = () => {
      if (!s.dead) {
        s.dead = true;
        s.term.write("\r\n[Verbindung getrennt]\r\n");
      }
      wsTermRenderList();
      wsTreeAutoRefreshUpdate();
    };
    // WICHTIG: Methoden-Form „term.onData(cb)“ — KEINE Property-Zuweisung!
    // In dieser xterm-Version ist Terminal.onData (wie onResize) ein
    // getter-only-Accessor ohne Setter: „term.onData = fn“ schlägt in
    // Non-Strict-JS stillfehend fehl und der Handler wird nie registriert.
    s.term.onData((data) => {
      if (s.ws.readyState !== WebSocket.OPEN) return;
      try {
        s.ws.send(wsTermEncoder.encode(data));
      } catch (err) { /* ignore */ }
    });
    s.term.onResize((size) => {
      if (s.ws.readyState === WebSocket.OPEN) {
        s.ws.send(JSON.stringify({ type: "resize", cols: size.cols, rows: size.rows }));
      }
    });
    wsTerm.sessions.push(s);
    wsTermRenderList();
    wsTermSelect(s);
    requestAnimationFrame(() => { try { s.fit.fit(); } catch (err) { /* ignore */ } });
    wsTreeAutoRefreshUpdate();
  }

  // ─── Terminal-Input-Fallback (Focus-unabhängig) ─────────────
  // xterms term.onData feuert NUR, wenn das versteckte Hilfs-Textarea
  // (xterm-helper-textarea) den DOM-Fokus hat — der geht hier leicht
  // verloren (Klicks auf Buttons/Body/Baum) → ohne Fallback erscheint
  // nichts im PTY. Solange die Terminal-View aktiv ist und der Fokus
  // auf einem „harmlosen" Element liegt, werden die Tasten hier an
  // die aktive (sichtbare) Session weitergeleitet. Liegt der Fokus
  // auf xterms Textarea, ist der xterm-eigene onData-Weg zuständig
  // (→ hier überspringen, keine doppelte Eingabe).
  const WS_TERM_KEYS = {
    Enter: "\r", Backspace: "\x7f", Tab: "\t", Escape: "\x1b",
    ArrowUp: "\x1b[A", ArrowDown: "\x1b[B",
    ArrowRight: "\x1b[C", ArrowLeft: "\x1b[D",
    Home: "\x1b[H", End: "\x1b[F", Delete: "\x1b[3~",
    PageUp: "\x1b[5~", PageDown: "\x1b[6~",
  };
  function wsTermSendText(s, text) {
    if (!s || !s.ws || s.ws.readyState !== WebSocket.OPEN || s.dead) return;
    try {
      s.ws.send(wsTermEncoder.encode(text));
    } catch (err) { /* ignore */ }
  }
  function wsTermEditableTarget(t) {
    if (!t || !t.closest) return false;
    return !!(t.isContentEditable ||
              t.closest("input, textarea, select") ||
              (t.classList && t.classList.contains("xterm-helper-textarea")));
  }
  function wsTermKeyFallback(e) {
    if (wsView !== "terminal" || !wsTerm.active || wsTerm.active.dead) return;
    if (e.metaKey || e.defaultPrevented) return;  // Browser-Shortcuts (Ctrl+S = workspace.js-Save)
    if (wsTermEditableTarget(e.target)) return;   // xterm-Weg bzw. echte Eingabefelder
    if (document.querySelector(".ws-ctx-menu")) return;
    let data = null;
    if (e.ctrlKey) {
      const k = (e.key || "").toLowerCase();
      if (k === "v") return;  // → paste-Event (wsTermPasteFallback)
      if (k.length === 1 && k >= "a" && k <= "z") {
        data = String.fromCharCode(k.charCodeAt(0) - 96);  // Ctrl+C → \x03 …
      } else if (e.key === "Enter") {
        data = "\r";
      }
    } else if (e.altKey && e.key.length === 1) {
      data = "\x1b" + e.key;  // Alt+Leiter → ESC-Präfix
    } else if (e.key in WS_TERM_KEYS) {
      data = WS_TERM_KEYS[e.key];
    } else if (e.key.length === 1) {
      data = e.key;
    }
    if (data === null) return;
    e.preventDefault();
    wsTermSendText(wsTerm.active, data);
  }
  function wsTermPasteFallback(e) {
    if (wsView !== "terminal" || !wsTerm.active || wsTerm.active.dead) return;
    if (wsTermEditableTarget(e.target)) return;
    if (document.querySelector(".ws-ctx-menu")) return;
    const cd = e.clipboardData || window.clipboardData;
    const text = cd ? cd.getData("text") : "";
    if (!text) return;
    e.preventDefault();
    wsTermSendText(wsTerm.active, text);
  }
  document.addEventListener("keydown", wsTermKeyFallback);
  document.addEventListener("paste", wsTermPasteFallback);

  // Während eines Runs morphen der gestartete Button in einen Stoppen-Button

  function wsMorphToStop(btn) {
    if (!btn) return;

    btn.dataset.origClass = btn.className;

    btn.className = "bg-red-600 text-white px-4 py-2 rounded-lg hover:bg-red-700 transition flex items-center gap-2";

  function wsUnmorph() {
    if (!wsStopMorphBtn) return;
    const btn = wsStopMorphBtn;
    wsStopMorphBtn = null;
    if (btn.dataset.origClass) btn.className = btn.dataset.origClass;
    if (btn.dataset.origHtml != null) btn.innerHTML = btn.dataset.origHtml;
    delete btn.dataset.origClass;
    delete btn.dataset.origHtml;
  }

  // kind: "run" (run.sh) | "test" (tests/test_public.sh) — Studenten senden
  // keine freien Commands, das Backend mappt auf die Skript-Befehle.
  async function wsRun(kind, srcBtn) {
    if (wsState.activeRun) return;  // laufender Job → keine parallelen Runs
    if (wsUI.isDirty()) {
      await wsUI.saveFile();  // aktuelle Datei vor dem Lauf speichern
    }
    const terminal = document.getElementById("ws-terminal");
    const termText = document.getElementById("ws-terminal-text");

    setRunButtonsDisabled(true);
    terminal.classList.remove("hidden");
    termText.textContent = "Lauf wird gestartet …";
    termText.style.color = "#9ca3af";

    try {
      const res = await fetch(WS_API + "/run", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "same-origin",
        body: JSON.stringify({ kind: kind || "run" }),
      });
      const data = await res.json();
      if (!res.ok) throw new Error(data.detail || res.status);

      if (data.async) {
        wsStartPolling(data.run_id, srcBtn);
      } else {
        wsRenderRunResult(data);
        setRunButtonsDisabled(false);
        wsUI.refresh();  // neue Artefakte (z. B. Plots) im Datei-Tree zeigen
      }
    } catch (err) {
      termText.textContent = "Fehler: " + err.message;
      termText.style.color = "#f87171";
      setRunButtonsDisabled(false);
    }
  }

  function wsStartPolling(runId, stopBtn) {
    wsState.activeRun = { runId, startedAt: Date.now() };
    wsMorphToStop(stopBtn);
    document.getElementById("ws-run-duration").textContent = "⏳ läuft seit 00:00";

    const tick = setInterval(() => {
      const s = Math.floor((Date.now() - wsState.activeRun.startedAt) / 1000);
      const mm = String(Math.floor(s / 60)).padStart(2, "0");
      const ss = String(s % 60).padStart(2, "0");
      document.getElementById("ws-run-duration").textContent =
        "⏳ läuft seit " + mm + ":" + ss;
    }, 1000);

    const poll = async () => {
      if (!wsState.activeRun || wsState.activeRun.runId !== runId) return;
      try {
        const res = await fetch(WS_API + "/runs/" + runId, { credentials: "same-origin" });
        const st = await res.json();
        if (st.status === "running" || st.status === "queued") {
          // Live-Log-Tail (Backend liefert stdout/stderr als String oder Array)
          const norm = (v) => Array.isArray(v) ? v.join("\n") : String(v || "");
          const out = norm(st.stdout);
          const err = norm(st.stderr);
          const combined = [out, err].filter(Boolean).join("\n");
          const termText = document.getElementById("ws-terminal-text");
          if (combined) {
            termText.textContent = combined;
            termText.style.color = "#86efac";
            termText.scrollTop = termText.scrollHeight;
          } else {
            termText.textContent = "⏳ Läuft … (noch keine Ausgabe)";
            termText.style.color = "#9ca3af";
          }
          if (st.status === "queued") {
            let qText = "⏳ wartet auf Start (GPU-Queue) …";
            if (st.queue_position != null) {
              qText = "⏳ wartet auf GPU (" + st.queue_position + " vor dir) …";
            }
            document.getElementById("ws-run-duration").textContent = qText;
          }
          wsState.activeRun.timer = setTimeout(poll, 2500);
        } else {
          // Fertig
          clearInterval(tick);
          const s = Math.floor((Date.now() - wsState.activeRun.startedAt) / 1000);
          const mm = String(Math.floor(s / 60)).padStart(2, "0");
          const ss = String(s % 60).padStart(2, "0");
          wsState.activeRun = null;
          wsUnmorph();
          wsRenderRunResult({
            status: st.status,
            exit_code: st.exit_code,
            stdout: st.stdout,
            stderr: st.stderr,
            timed_out: st.status === "timeout",
            duration: st.duration,
          });
          document.getElementById("ws-run-duration").textContent = "⏱ " + mm + ":" + ss;
          setRunButtonsDisabled(false);
          wsUI.refresh();  // neue/geänderte Dateien im Datei-Tree zeigen
          wsLoadStatus();  // Banner + Disk-Anzeige (z. B. Quota-Nachlauf) aktualisieren
        }
      } catch (err) {
        // Agent kurz down? Weiter pollen (Backend fällt auf DB-Historie zurück).
        wsState.activeRun.timer = setTimeout(poll, 5000);
      }
    };
    wsState.activeRun.timer = setTimeout(poll, 2000);
  }

  async function wsStopRun() {
    if (!wsState.activeRun) return;
    const runId = wsState.activeRun.runId;
    try {
      const res = await fetch(WS_API + "/runs/" + runId + "/stop", {
        method: "POST", credentials: "same-origin",
      });
      if (!res.ok && res.status !== 404) {
        const data = await res.json();
        showToast("Stoppen fehlgeschlagen: " + (data.detail || res.status), "error");
      }
    } catch (err) {
      showToast("Stoppen fehlgeschlagen: " + err.message, "error");
    }
  }

  function wsRenderRunResult(data) {
    const termText = document.getElementById("ws-terminal-text");
    let output = "";
    if (data.stdout && String(data.stdout).trim()) output += String(data.stdout);
    if (data.stderr && String(data.stderr).trim()) {
      if (output) output += "\n\n";
      output += "--- FEHLER ---\n" + data.stderr.trim();
    }
    if (!output.trim()) output = "(Keine Ausgabe)";
    if (data.timed_out || data.status === "timeout") {
      output += "\n\n[⏰ TIMEOUT: Zeitlimit erreicht]";
    }
    termText.textContent = output;
    const failed = (data.exit_code != null && data.exit_code !== 0) || data.timed_out;
    termText.style.color = (failed && data.stderr) ? "#f87171" : "#86efac";
    termText.scrollTop = termText.scrollHeight;

    if (data.exit_code != null) {
      document.getElementById("ws-run-duration").textContent =
        (data.duration ? "⏱ " + Math.round(data.duration) + " s · " : "") +
        "Exit-Code: " + data.exit_code;
    }
  }
